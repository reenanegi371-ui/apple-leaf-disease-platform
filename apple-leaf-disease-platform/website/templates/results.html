class ApiService { static const String baseUrl = 'http://192.168.1.100:8000'; // Replace with your backend IP static Future checkHealth() async { try { final response = await http.get( Uri.parse('$baseUrl/health'), headers: {'Content-Type': 'application/json'}, ).timeout(Duration(seconds: 5)); return response.statusCode == 200; } catch (e) { print('Health check failed: $e'); return false; } } static Future detectDisease(File imageFile) async { try { var request = http.MultipartRequest( 'POST', Uri.parse('$baseUrl/detect'), );
